    30→/* ============================================================
    31→   Counter Hook
    32→   ============================================================ */
    33→
    34→function useCounter(end: number, duration = 2000, shouldStart: boolean) {
    35→  const [count, setCount] = useState(0);
    36→
    37→  useEffect(() => {
    38→    if (!shouldStart) return;
    39→    let startTime: number | null = null;
    40→    let raf: number;
    41→    const step = (timestamp: number) => {
    42→      if (!startTime) startTime = timestamp;
    43→      const progress = Math.min((timestamp - startTime) / duration, 1);
    44→      const eased = 1 - Math.pow(1 - progress, 3);
    45→      setCount(Math.floor(eased * end * 10) / 10);
    46→      if (progress < 1) raf = requestAnimationFrame(step);
    47→    };
    48→    raf = requestAnimationFrame(step);
    49→    return () => cancelAnimationFrame(raf);
    50→  }, [shouldStart, end, duration]);
    51→
    52→  return count;
    53→}
