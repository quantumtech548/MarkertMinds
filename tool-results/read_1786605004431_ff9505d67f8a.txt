    30→/* ============================================================
    31→   Counter Hook
    32→   ============================================================ */
    33→
    34→function useCounter(end: number, duration = 2000, startOnView = true) {
    35→  const [count, setCount] = useState(0);
    36→  const [started, setStarted] = useState(!startOnView);
    37→  const ref = useRef<HTMLDivElement>(null);
    38→
    39→  useEffect(() => {
    40→    if (!startOnView) return;
    41→    const el = ref.current;
    42→    if (!el) return;
    43→    const observer = new IntersectionObserver(
    44→      ([entry]) => {
    45→        if (entry.isIntersecting) {
    46→          setStarted(true);
    47→          observer.disconnect();
    48→        }
    49→      },
    50→      { threshold: 0.3 }
    51→    );
    52→    observer.observe(el);
    53→    return () => observer.disconnect();
    54→  }, [startOnView]);
    55→
    56→  useEffect(() => {
    57→    if (!started) return;
    58→    let startTime: number | null = null;
    59→    let raf: number;
    60→    const step = (timestamp: number) => {
    61→      if (!startTime) startTime = timestamp;
    62→      const progress = Math.min((timestamp - startTime) / duration, 1);
    63→      const eased = 1 - Math.pow(1 - progress, 3);
    64→      setCount(Math.floor(eased * end));
    65→      if (progress < 1) raf = requestAnimationFrame(step);
    66→    };
    67→    raf = requestAnimationFrame(step);
    68→    return () => cancelAnimationFrame(raf);
    69→  }, [started, end, duration]);
    70→
    71→  return { count, ref };
    72→}
